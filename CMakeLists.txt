cmake_minimum_required(VERSION 3.20)
project(synthgen-core LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)

find_package(Arrow REQUIRED)
find_package(Parquet REQUIRED)
find_package(GTest REQUIRED)

add_subdirectory(src/common)
add_subdirectory(src/parser)
add_subdirectory(src/schema)
add_subdirectory(src/engine/physics)
add_subdirectory(src/engine/constraint)
add_subdirectory(src/engine/evidence)
add_subdirectory(src/engine/router)
add_subdirectory(src/engine/postfilter)
add_subdirectory(src/storage)
add_subdirectory(src/storage/audit)
add_subdirectory(src/storage/version)
add_subdirectory(src/storage/gc)
add_subdirectory(src/storage/timetravel)
add_subdirectory(src/storage/model)
add_subdirectory(src/engine/alignment)
add_subdirectory(src/scaffold)
add_subdirectory(src/api)
add_subdirectory(src/sdk)
add_subdirectory(tools)

enable_testing()
add_subdirectory(tests)
